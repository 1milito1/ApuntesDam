import java.io.*;
import java.net.*;
import java.util.Random;

public class Servidor {
    public static void main(String[] args) {
        int puerto = 12345; // Puerto del servidor
        try (ServerSocket serverSocket = new ServerSocket(puerto)) {
            System.out.println("Servidor iniciado en el puerto " + puerto);
            while (true) {
                Socket socketCliente = serverSocket.accept();
                System.out.println("Cliente conectado: " + socketCliente.getInetAddress());
                new HiloCliente(socketCliente).start();
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}

class HiloCliente extends Thread {
    private Socket socket;

    public HiloCliente(Socket socket) {
        this.socket = socket;
    }

    public void run() {
        try (BufferedReader entrada = new BufferedReader(new InputStreamReader(socket.getInputStream()));
             PrintWriter salida = new PrintWriter(socket.getOutputStream(), true)) {

            String opcion;
            do {
                // Enviar opciones al cliente
                salida.println("¿Qué quieres hacer? [Codigo | Password | Navegador]");
                salida.println("Escribe Exit para terminar la conexión");

                // Leer la opción del cliente
                opcion = entrada.readLine();

                if (opcion == null) {
                    System.out.println("El cliente ha cerrado la conexión.");
                    break;
                }

                switch (opcion.toLowerCase()) {
                    case "codigo":
                        enviarContenidoArchivo(salida);
                        break;
                    case "password":
                        generarPassword(entrada, salida);
                        break;
                    case "navegador":
                        abrirNavegador();
                        salida.println("Navegador abierto.");
                        break;
                    case "exit":
                        salida.println("Conexión terminada.");
                        break;
                    default:
                        salida.println("Opción no válida. Intenta de nuevo.");
                }
            } while (!opcion.equalsIgnoreCase("exit"));
        } catch (IOException e) {
            System.err.println("Error en la conexión con el cliente: " + e.getMessage());
        } finally {
            try {
                socket.close();
            } catch (IOException e) {
                System.err.println("Error al cerrar el socket: " + e.getMessage());
            }
        }
    }

    private void enviarContenidoArchivo(PrintWriter salida) {
        try (BufferedReader br = new BufferedReader(new FileReader("codigo.txt"))) {
            String linea;
            StringBuilder contenido = new StringBuilder();
            while ((linea = br.readLine()) != null) {
                contenido.append(linea).append("\n");
            }
            salida.println("Contenido del archivo:\n" + contenido.toString());
        } catch (IOException e) {
            salida.println("Error al leer el archivo.");
        }
    }

    private void generarPassword(BufferedReader entrada, PrintWriter salida) throws IOException {
        salida.println("Introduce la longitud de la password:");
        String longitudStr = entrada.readLine();
        try {
            int longitud = Integer.parseInt(longitudStr);
            String password = generarPasswordAleatoria(longitud);
            salida.println("Password generada: " + password);
        } catch (NumberFormatException e) {
            salida.println("Longitud no válida. Debe ser un número entero.");
        }
    }

    private String generarPasswordAleatoria(int longitud) {
        String caracteres = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
        Random random = new Random();
        StringBuilder password = new StringBuilder(longitud);
        for (int i = 0; i < longitud; i++) {
            int indice = random.nextInt(caracteres.length());
            password.append(caracteres.charAt(indice));
        }
        return password.toString();
    }

    private void abrirNavegador() {
        try {
            Runtime.getRuntime().exec("firefox");
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}



import java.io.*;
import java.net.*;

public class Cliente {
    public static void main(String[] args) {
        String host = "localhost"; // Cambia esto si el servidor está en otra máquina
        int puerto = 12345; // Debe coincidir con el puerto del servidor

        try (Socket socket = new Socket(host, puerto);
             BufferedReader entrada = new BufferedReader(new InputStreamReader(socket.getInputStream()));
             PrintWriter salida = new PrintWriter(socket.getOutputStream(), true);
             BufferedReader teclado = new BufferedReader(new InputStreamReader(System.in))) {

            String respuesta;
            do {
                // Leer y mostrar la respuesta del servidor
                respuesta = entrada.readLine();
                if (respuesta != null) {
                    System.out.println(respuesta);
                }

                // Si la respuesta no es "Conexión terminada", solicitar la opción al usuario
                if (!respuesta.equals("Conexión terminada.")) {
                    // Mostrar "Elige una opción:" solo si no se está pidiendo la longitud
                    if (!respuesta.startsWith("Introduce la longitud de la password:")) {
                        System.out.print("Elige una opción: ");
                        String opcion = teclado.readLine();
                        salida.println(opcion);

                        // Si el cliente elige "Password", esperar la longitud
                        if (opcion.equalsIgnoreCase("password")) {
                            respuesta = entrada.readLine(); // Leer la solicitud de longitud
                            System.out.println(respuesta);
                            System.out.print("Introduce la longitud de la password: ");
                            String longitudStr = teclado.readLine();
                            salida.println(longitudStr);
                        }
                    }
                }
            } while (respuesta != null && !respuesta.equals("Conexión terminada."));
        } catch (IOException e) {
            System.err.println("Error en la conexión con el servidor: " + e.getMessage());
        }
    }
}
