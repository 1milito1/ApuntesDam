Creación de un Menú y Manejo de Elección de Casos, Lectura y Escritura de Archivos en Java

import java.util.Scanner;

public class MenuExample {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int opcion;

        do {
            System.out.println("Menú de Opciones:");
            System.out.println("1. Saludar");
            System.out.println("2. Sumar dos números");
            System.out.println("3. Leer un archivo");
            System.out.println("4. Escribir en un archivo");
            System.out.println("5. Salir");
            System.out.print("Elige una opción: ");
            opcion = scanner.nextInt();
            scanner.nextLine();  // Limpiar el buffer

            switch (opcion) {
                case 1:
                    saludar(scanner);
                    break;
                case 2:
                    sumar(scanner);
                    break;
                case 3:
                    leerArchivo();
                    break;
                case 4:
                    escribirArchivo(scanner);
                    break;
                case 5:
                    System.out.println("Saliendo...");
                    break;
                default:
                    System.out.println("Opción no válida. Inténtalo de nuevo.");
            }
        } while (opcion != 5);
    }

    public static void saludar(Scanner scanner) {
        System.out.print("Ingresa tu nombre: ");
        String nombre = scanner.nextLine();
        System.out.println("Hola, " + nombre + "!");
    }

    public static void sumar(Scanner scanner) {
        System.out.print("Ingresa el primer número: ");
        int num1 = scanner.nextInt();
        System.out.print("Ingresa el segundo número: ");
        int num2 = scanner.nextInt();
        int suma = num1 + num2;
        System.out.println("La suma es: " + suma);
    }

    public static void leerArchivo() {
        try {
            java.nio.file.Path path = java.nio.file.Paths.get("archivo.txt");
            java.util.List<String> lines = java.nio.file.Files.readAllLines(path);
            System.out.println("Contenido del archivo:");
            for (String line : lines) {
                System.out.println(line);
            }
        } catch (java.io.IOException e) {
            System.out.println("Ocurrió un error al leer el archivo.");
            e.printStackTrace();
        }
    }

    public static void escribirArchivo(Scanner scanner) {
        try {
            java.nio.file.Path path = java.nio.file.Paths.get("archivo.txt");
            System.out.println("Escribe el contenido que deseas agregar al archivo:");
            String contenido = scanner.nextLine();
            java.nio.file.Files.write(path, contenido.getBytes(), java.nio.file.StandardOpenOption.CREATE, java.nio.file.StandardOpenOption.APPEND);
            System.out.println("Contenido escrito en el archivo.");
        } catch (java.io.IOException e) {
            System.out.println("Ocurrió un error al escribir en el archivo.");
            e.printStackTrace();
        }
    }
}


Uso de HashMap en Java y Funciones del Menú

1. Ejemplo Básico de HashMap:
import java.util.HashMap;
import java.util.Map;

public class HashMapExample {
    public static void main(String[] args) {
        HashMap<String, Integer> edades = new HashMap<>();
        edades.put("Juan", 25);
        edades.put("María", 30);
        edades.put("Pedro", 35);
        int edadJuan = edades.get("Juan");
        System.out.println("La edad de Juan es: " + edadJuan);
        if (edades.containsKey("María")) {
            System.out.println("María está en el HashMap.");
        }
        if (edades.containsValue(35)) {
            System.out.println("Alguien tiene 35 años.");
        }
        edades.remove("Pedro");
        for (Map.Entry<String, Integer> entrada : edades.entrySet()) {
            System.out.println("Clave: " + entrada.getKey() + ", Valor: " + entrada.getValue());
        }
    }
}

2. Funciones del Menú para Usar HashMap:
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class MenuHashMapExample {
    private static HashMap<String, Integer> personas = new HashMap<>();
    private static Scanner scanner = new Scanner(System.in);

    public static void main(String[] args) {
        int opcion;

        do {
            System.out.println("Menú de Opciones:");
            System.out.println("1. Agregar persona");
            System.out.println("2. Buscar edad por nombre");
            System.out.println("3. Mostrar todas las personas");
            System.out.println("4. Eliminar persona");
            System.out.println("5. Salir");
            System.out.print("Elige una opción: ");
            opcion = scanner.nextInt();
            scanner.nextLine();  // Limpiar el buffer

            switch (opcion) {
                case 1:
                    agregarPersona();
                    break;
                case 2:
                    buscarEdad();
                    break;
                case 3:
                    mostrarPersonas();
                    break;
                case 4:
                    eliminarPersona();
                    break;
                case 5:
                    System.out.println("Saliendo...");
                    break;
                default:
                    System.out.println("Opción no válida. Inténtalo de nuevo.");
            }
        } while (opcion != 5);
    }

    public static void agregarPersona() {
        System.out.print("Ingresa el nombre: ");
        String nombre = scanner.nextLine();
        System.out.print("Ingresa la edad: ");
        int edad = scanner.nextInt();
        scanner.nextLine();  // Limpiar el buffer
        personas.put(nombre, edad);
        System.out.println("Persona agregada.");
    }

    public static void buscarEdad() {
        System.out.print("Ingresa el nombre: ");
        String nombre = scanner.nextLine();
        if (personas.containsKey(nombre)) {
            int edad = personas.get(nombre);
            System.out.println(nombre + " tiene " + edad + " años.");
        } else {
            System.out.println("No se encontró a " + nombre + " en el registro.");
        }
    }

    public static void mostrarPersonas() {
        if (personas.isEmpty()) {
            System.out.println("No hay personas en el registro.");
        } else {
            for (Map.Entry<String, Integer> entrada : personas.entrySet()) {
                System.out.println("Nombre: " + entrada.getKey() + ", Edad: " + entrada.getValue());
            }
        }
    }

    public static void eliminarPersona() {
        System.out.print("Ingresa el nombre: ");
        String nombre = scanner.nextLine();
        if (personas.containsKey(nombre)) {
            personas.remove(nombre);
            System.out.println(nombre + " ha sido eliminado del registro.");
        } else {
            System.out.println("No se encontró a " + nombre + " en el registro.");
        }
    }
}

public class Contacto implements Comparable<Contacto> {
    private String nombre;
    private String apellidos;
    private String email;

    public Contacto(String nombre, String apellidos, String email) {
        this.nombre = nombre;
        this.apellidos = apellidos;
        this.email = email;
    }

    public String getNombre() {
        return nombre;
    }

    public String getApellidos() {
        return apellidos;
    }

    public String getEmail() {
        return email;
    }

    @Override
    public String toString() {
        return "Contacto{" +
                "nombre='" + nombre + '\'' +
                ", apellidos='" + apellidos + '\'' +
                ", email='" + email + '\'' +
                '}';
    }

    @Override
    public int compareTo(Contacto otro) {
        return this.email.compareTo(otro.email);
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;

        Contacto contacto = (Contacto) o;

        return email.equals(contacto.email);
    }

    @Override
    public int hashCode() {
        return email.hashCode();
    }
}


clase csvreader

import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;
import java.util.TreeSet;

public class CSVReader {
    public static TreeSet<Contacto> leerContactosDesdeCSV(String archivo) {
        TreeSet<Contacto> contactos = new TreeSet<>();
        try (BufferedReader br = new BufferedReader(new FileReader(archivo))) {
            String linea;
            br.readLine(); // Saltar la primera línea si contiene encabezados
            while ((linea = br.readLine()) != null) {
                String[] campos = linea.split(",");
                if (campos.length > 30) {
                    String nombre = campos[1];
                    String apellidos = campos[3];
                    String email = campos[30];
                    Contacto contacto = new Contacto(nombre, apellidos, email);
                    contactos.add(contacto);
                }
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
        return contactos;
    }
}

public class ContactoWriter {
    public static void escribirContactosAArchivo(Set<Contacto> contactos, String archivoSalida) {
        try (BufferedWriter bw = new BufferedWriter(new FileWriter(archivoSalida))) {
            for (Contacto contacto : contactos) {
                bw.write(contacto.toString());
                bw.newLine();
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}

import javax.swing.JFrame;
import javax.swing.JLabel;
import javax.swing.JButton;
import javax.swing.JPanel;
import java.awt.BorderLayout;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;

public class SwingExample {
    public static void main(String[] args) {
        // Crear una nueva ventana
        JFrame frame = new JFrame("Ejemplo de Swing");
        frame.setSize(400, 200);
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);

        // Crear un panel para contener componentes
        JPanel panel = new JPanel();
        frame.add(panel);
        placeComponents(panel);

        // Hacer visible la ventana
        frame.setVisible(true);
    }

    private static void placeComponents(JPanel panel) {
        panel.setLayout(null);

        // Crear una etiqueta
        JLabel userLabel = new JLabel("Usuario:");
        userLabel.setBounds(10, 20, 80, 25);
        panel.add(userLabel);

        // Crear un botón
        JButton loginButton = new JButton("Iniciar sesión");
        loginButton.setBounds(10, 80, 150, 25);
        panel.add(loginButton);

        // Añadir acción al botón
        loginButton.addActionListener(new ActionListener() {
            @Override
            public void actionPerformed(ActionEvent e) {
                System.out.println("Botón de inicio de sesión presionado");
            }
        });
    }
}

import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class JavaFXExample extends Application {
    @Override
    public void start(Stage primaryStage) {
        // Crear una etiqueta
        Label label = new Label("Hola, JavaFX!");

        // Crear un botón
        Button button = new Button("Presionar");
        button.setOnAction(e -> System.out.println("Botón presionado"));

        // Crear un contenedor (VBox)
        VBox vbox = new VBox(10, label, button);

        // Crear una escena y añadir el contenedor
        Scene scene = new Scene(vbox, 300, 200);

        // Configurar el escenario (ventana)
        primaryStage.setTitle("Ejemplo de JavaFX");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
