    private static void nuevoServicio(Servidor servidor, Scanner scanner) {
        System.out.println("\n--- Crear Nuevo Servicio ---");
        System.out.println("Tipo de serivicio (1= Archivo, 2=Alojamienot, 3 Aplicaciones):");
        int tipo = scanner.nextInt();
        scanner.nextLine();
        
        System.out.println("Nombre del servicio: ");
        String nombre = scanner.nextLine();
        
        System.out.println("Precio base: ");
        double precio = scanner.nextDouble();
        
        Servicio servicio = null;
        
        switch (tipo) {
            case 1:
                System.out.println("Almacenamiento (GB): ");
                int almacenamiento = scanner.nextInt();
                System.out.println("Coste por GB de almacenamiento: ");
                double costePorGBAlmacenamiento = scanner.nextDouble();
                servicio = new archivo(nombre, precio, almacenamiento, costePorGBAlmacenamiento);
                break;

                System.out.println("Máximo de sitios web: ");
                int maxSitiosWeb = scanner.nextInt();
                System.out.println("coste por sitio: ");
                double costePorSitio = scanner.nextDouble();
                servicio = new Alojamiento(nombre, precio, maxSitiosWeb, costePorSitio);
                break;
            case 3:
                System.out.println("Almacenamiento (G): ");
                int alm = scanner.nextInt();
                System.out.println("Coste por GB de almacenamiento: ");
                double costeAlm = scanner.nextDouble();
                System.out.println("RAM (GB): ");
                int ram = scanner.nextInt();
                System.out.println("Coste por GB de RAM: ");
                double costePorGBRam = scanner.nextDouble();
                servicio = new Aplicaciones (nombre, precio, alm, costeAlm, ram, costePorGBRam);
                break;
            default:
                System.out.println("Tipo de servicio no valido");
                return;
        }
        
        if (servidor.agregarServicio(servicio)){
            System.out.println("Servicio añadido exitosamente.");
        }else {
            System.out.println("El servicio ya existe y no se puede duplicar.");
        }
    }

    private static void listadoServicios(Servidor servidor, Scanner scanner) {
        System.out.println("\n --- Listado de servicios ---");
        servidor.mostrarServicios();
        
        double totalFacturacion = servidor.getTotalFacturacion();
        System.out.println("\nTotal de facturacion: " + totalFacturacion + " €");
 

    }

    private static void borrarServicio(Servidor servidor, Scanner scanner) {
        System.out.println("\n Nombre del servicio a borrar: ");
        String nombre = scanner.nextLine();
        
        if (servidor.eliminarServicioPorNombre(nombre)){
            System.out.println("Servicio eliminado");
        }else {
            System.out.println("Servicio no encontrado");
        }
    }

    private static void crearResumen(Servidor servidor) {
        System.out.println("\n--- Resumen de Servicios ---");
        List<ResumenServicio> resumen = servidor.crearResumenServicios();
        
        resumen.sort(Comparator.comparing(ResumenServicio::getNombre));
        
        for(ResumenServicio rs : resumen){
            System.out.println(rs);
        }
        double totalFacturacion = servidor.getTotalFacturacion();
        System.out.println("\nTotal de facturación; " + totalFacturacion + " €");
    }

    private static void opcionAElegir(Servidor servidor, Scanner scanner) {
        
        System.out.println("\nElige una opcion:");
        System.out.println("1. listado grafico");
        System.out.println("2. Guardar en fichero.");
        int opcion = scanner.nextInt();
        
        switch (opcion) {
            case 1:
                listadorGrafico(servidor);
                
                break;
            case 2:
                guardarFichero(servidor);
                break;
            default:
                System.out.println("Opcion no valida");
        }

    }

    private static void listadorGrafico(Servidor servidor) {
        
       StringBuilder mensaje = new StringBuilder("Listado de servicios:\n");
       for(Servicio servicio : servidor.getServicios()){
           mensaje.append(servicio).append("\n");
       }
       
       JOptionPane.showMessageDialog(null, mensaje.toString(), "Listado de servicios", JOptionPane.INFORMATION_MESSAGE);
    }

    private static void guardarFichero(Servidor servidor) {
        try (BufferedWriter writer = new BufferedWriter(new FileWriter("servicios.txt"))) {
            for (Servicio servicio : servidor.getServicios()){
                writer.write(servicio.toString());
                writer.newLine();
            }
            System.out.println("Servicio guardados en el fichero servicios.txt");
        } catch (IOException e) {
            System.out.println("error al guardar en el fichero:" + e.getMessage());
        }
 
        
    }
    
    
    
        
    
    
}

public class Servicio {
    
    private String nombre;
    private double precio;
    
    public Servicio(){
        this.nombre = "Sin nombre";
        this.precio = 0.0;
    }

    public Servicio(String nombre, double precio) {
        this.nombre = nombre;
        this.precio = precio;
    }

    public String getNombre() {
        return nombre;
    }

    public void setNombre(String nombre) {
        this.nombre = nombre;
    }

    public double getPrecio() {
        return precio;
    }

    public void setPrecio(double precio) {
        this.precio = precio;
    }

    @Override
    public String toString() {
        return "Servicio{" + "nombre=" + nombre + ", precio=" + precio + '}';
    }
    
    public double calcularPrecio(){
        return precio;
    }
}

/**

 * @author milio
 */
public class archivo extends Servicio {
    
    private int almacenamiento;
    private double costePorGBAlmacenamiento;
    
    public archivo(){
        super("servicio sin nombre", 0.0);
        this.almacenamiento = 0;
        this.costePorGBAlmacenamiento = 0.0;
    }

    public archivo(String nombre, double precio, int almacenamiento, double costePorGBAlmacenamiento) {
        super(nombre, precio);
        this.almacenamiento = almacenamiento;
        this.costePorGBAlmacenamiento = costePorGBAlmacenamiento;
    }

    public int getAlmacenamiento() {
        return almacenamiento;
    }

    public void setAlmacenamiento(int almacenamiento) {
        this.almacenamiento = almacenamiento;
    }

    public double getCostePorGBAlmacenamiento() {
        return costePorGBAlmacenamiento;
    }

    public void setCostePorGBAlmacenamiento(double costePorGBAlmacenamiento) {
        this.costePorGBAlmacenamiento = costePorGBAlmacenamiento;
    }
    
    @Override
    public double calcularPrecio(){
        return super.getPrecio() + (almacenamiento * costePorGBAlmacenamiento);
    }

    @Override
    public String toString() {
        return "archivo{" + "almacenamiento=" + almacenamiento + ", costePorGBAlmacenamiento=" + costePorGBAlmacenamiento + ", precio total=" + calcularPrecio()+ '}';
    }
    
}

package parcial2;

import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Set;

/**
 *
 * @author milio
 */
public class Servidor {
    
    private Set<Servicio> servicios;
    
    public Servidor(){
        this.servicios = new LinkedHashSet<>();
    }
    
    public boolean agregarServicio(Servicio servicio){
        return servicios.add(servicio);
    }
    
    public Servicio buscarServicioPorNombre(String nombre){
        for(Servicio servicio : servicios){
            if(servicio.getNombre().equalsIgnoreCase(nombre)){
                return servicio;
            }
        }
        return null;
    }
    
    public boolean eliminarServicioPorNombre(String nombre){
        return servicios.removeIf(servicio -> servicio.getNombre().equalsIgnoreCase(nombre));
    }
    
    public void mostrarServicios(){
        for (Servicio servicio : servicios){
            System.out.println(servicio);
        }
    }
    
    public double getTotalFacturacion(){
        double total = 0.0;
        for (Servicio servicio : servicios){
            total += servicio.calcularPrecio();
        }
        return total;
    }
    
    public List<ResumenServicio> crearResumenServicios(){
        List<ResumenServicio> resumen = new ArrayList<>();
        for (Servicio servicio : servicios){
            resumen.add(new ResumenServicio(servicio.getNombre(), servicio.calcularPrecio()));
            
        }        
        return resumen;
    }
    public Set<Servicio> getServicios(){
        return servicios;
    }
}
