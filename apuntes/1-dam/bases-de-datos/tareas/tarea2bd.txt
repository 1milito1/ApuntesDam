-- Creación de la tabla PAÍS
CREATE TABLE PAIS (
    COD_PAIS CHAR(3) NOT NULL,
    NOMBRE_PAIS VARCHAR(9) NOT NULL,
    NRO_ESTADIOS INT,
    PRIMARY KEY (COD_PAIS)
);

-- Creación de la tabla ESTADIO
CREATE TABLE ESTADIO (
    COD_ESTADIO CHAR(6) PRIMARY KEY,
    NOMBRE VARCHAR(20),
    CAPACIDAD INT,
    PAIS_ESTADIO CHAR(3),
    FECHA_INICIO DATE,
    FECHA_FIN DATE,
    CUBIERTA VARCHAR(11),
    FOREIGN KEY (PAIS_ESTADIO) REFERENCES PAIS(COD_PAIS),
    CHECK (FECHA_FIN > FECHA_INICIO)
);

-- Creación de la tabla AGENTE
CREATE TABLE AGENTE (
    NIF CHAR(9) PRIMARY KEY,
    NOMBRE VARCHAR(20),
    APELLIDO1 VARCHAR(20),
    APELLIDO2 VARCHAR(20),
    SEXO CHAR(1) CHECK (SEXO IN ('H', 'M')),
    ESTADIO_AGENTE CHAR(6),
    FOREIGN KEY (ESTADIO_AGENTE) REFERENCES ESTADIO(COD_ESTADIO)
);
